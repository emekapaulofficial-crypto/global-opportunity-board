name: Update Job & Scholarship Listings

on:
  schedule:
    # Runs 4 times a day (00:00, 06:00, 12:00, 18:00 UTC) — no manual work needed
    - cron: "0 */6 * * *"
  workflow_dispatch: {}   # lets you also trigger it manually from the Actions tab

permissions:
  contents: write   # needed so the workflow can commit data/listings.json back

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - name: Check out repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: pip install requests beautifulsoup4

      - name: Run job scraper
        env:
          ADZUNA_APP_ID: ${{ secrets.ADZUNA_APP_ID }}
          ADZUNA_APP_KEY: ${{ secrets.ADZUNA_APP_KEY }}
        run: python scripts/fetch_jobs.py

      - name: Run scholarship scraper
        run: python scripts/fetch_scholarships.py

      # This is what makes it fully automatic with no Google Sheet: the
      # scrapers write straight into data/listings.json, and this step
      # commits + pushes that file back to the repo. Cloudflare Pages is
      # watching the repo, so that push alone triggers a redeploy — the
      # live site picks up the new listings within a minute or two, with
      # zero manual steps.
      - name: Commit updated listings
        run: |
          git config user.name "opportunity-desk-bot"
          git config user.email "actions@users.noreply.github.com"
          git add data/listings.json
          git diff --staged --quiet && echo "No new listings this run." || git commit -m "Auto-update listings [skip ci]"
          git push
